01 // ví dụ xuyên suốt

tinhHocPhi theo R3

function tinhHocPhi(tinChi, donGia, chinhSach, hocLai) {
  if (tinChi <= 0) {
    throw new Error('Số tín chỉ phải dương');
  }
  let tong = tinChi * donGia;
  if (hocLai > 0) {
    tong = tong + hocLai * donGia * 1.2;
  }
  if (chinhSach) {
    tong = tong * 0.5;
  }
  if (tong > 30000000) {
    tong = 30000000;
  }
  return Math.round(tong);
}
